cmake_minimum_required(VERSION 3.25)
project(PixelWorld VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
include(FetchContent)
include(CTest)
option(PIXEL_BUILD_GAME "Build the Vulkan desktop application" ON)
option(PIXEL_BUILD_SCRIPTS "Build the managed gameplay assembly" ON)

# Dependency revisions are fixed so machines build the same engine.
set(GLM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glm URL https://github.com/g-truc/glm/archive/refs/tags/1.0.1.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/archive/refs/tags/v3.11.3.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(glm json)
find_package(Threads REQUIRED)

add_library(pixel_core STATIC engine/src/World.cpp engine/src/Physics.cpp engine/src/Storage.cpp engine/src/ScriptHost.cpp)
target_include_directories(pixel_core PUBLIC engine/include)
target_link_libraries(pixel_core PUBLIC glm::glm nlohmann_json::nlohmann_json Threads::Threads ${CMAKE_DL_LIBS})
target_compile_definitions(pixel_core PUBLIC GLM_FORCE_DEPTH_ZERO_TO_ONE GLM_FORCE_RADIANS NOMINMAX)
if(MSVC)
    target_compile_options(pixel_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(pixel_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(PIXEL_BUILD_SCRIPTS)
    find_program(DOTNET_EXECUTABLE dotnet REQUIRED)
    add_custom_target(pixel_scripts ALL
        COMMAND "${DOTNET_EXECUTABLE}" build "${CMAKE_SOURCE_DIR}/scripts/PixelWorld.Scripts.csproj" -c Release -o "${CMAKE_BINARY_DIR}/bin/scripts" --nologo
        COMMENT "Building C# gameplay scripts" VERBATIM)
endif()

if(PIXEL_BUILD_GAME)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_Declare(vulkan_headers URL https://github.com/KhronosGroup/Vulkan-Headers/archive/d4a196d8c84e032d27f999adcea3075517c1c97f.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_Declare(volk URL https://github.com/zeux/volk/archive/0b17a763ba5643e32da1b2152f8140461b3b7345.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(glfw vulkan_headers)
    set(VOLK_PULL_IN_VULKAN OFF CACHE BOOL "" FORCE)
    set(VOLK_INSTALL OFF CACHE BOOL "" FORCE)
    set(VOLK_HEADERS_ONLY OFF CACHE BOOL "" FORCE)
    set(VULKAN_HEADERS_INSTALL_DIR "${vulkan_headers_SOURCE_DIR}")
    FetchContent_MakeAvailable(volk)
    target_link_libraries(volk PUBLIC Vulkan::Headers)

    FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/f5befd2d29e66809cd1110a152e375a7f1981f06.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_Declare(miniaudio URL https://github.com/mackron/miniaudio/archive/refs/tags/0.11.21.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(imgui miniaudio)
    add_library(pixel_imgui STATIC
        "${imgui_SOURCE_DIR}/imgui.cpp" "${imgui_SOURCE_DIR}/imgui_draw.cpp"
        "${imgui_SOURCE_DIR}/imgui_widgets.cpp" "${imgui_SOURCE_DIR}/imgui_tables.cpp"
        "${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp"
        "${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp")
    target_include_directories(pixel_imgui PUBLIC "${imgui_SOURCE_DIR}" "${imgui_SOURCE_DIR}/backends")
    target_link_libraries(pixel_imgui PUBLIC glfw volk)
    target_compile_definitions(pixel_imgui PUBLIC VK_NO_PROTOTYPES IMGUI_IMPL_VULKAN_USE_VOLK GLFW_INCLUDE_NONE)

    # Build Khronos' shader compiler if no SDK is installed. Runtime only needs a GPU driver.
    find_program(GLSLANG_EXECUTABLE NAMES glslangValidator glslang HINTS "$ENV{VULKAN_SDK}/Bin")
    if(NOT GLSLANG_EXECUTABLE)
        set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
        set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
        set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
        set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
        set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
        set(ENABLE_GLSLANG_BINARIES ON CACHE BOOL "" FORCE)
        FetchContent_Declare(glslang URL https://github.com/KhronosGroup/glslang/archive/1062752a891c95b2bfeed9e356562d88f9df84ac.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
        FetchContent_MakeAvailable(glslang)
        set(GLSLANG_EXECUTABLE $<TARGET_FILE:glslang-standalone>)
        set(PIXEL_SHADER_COMPILER_DEP glslang-standalone)
    endif()
    set(PIXEL_SHADER_OUTPUTS)
    foreach(SHADER terrain.vert terrain.frag sky.vert sky.frag)
        set(SHADER_OUT "${CMAKE_BINARY_DIR}/bin/assets/shaders/${SHADER}.spv")
        add_custom_command(OUTPUT "${SHADER_OUT}"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/bin/assets/shaders"
            COMMAND "${GLSLANG_EXECUTABLE}" -V --target-env vulkan1.1 "${CMAKE_SOURCE_DIR}/assets/shaders/${SHADER}" -o "${SHADER_OUT}"
            DEPENDS "${CMAKE_SOURCE_DIR}/assets/shaders/${SHADER}" ${PIXEL_SHADER_COMPILER_DEP} VERBATIM)
        list(APPEND PIXEL_SHADER_OUTPUTS "${SHADER_OUT}")
    endforeach()
    add_custom_target(pixel_shaders ALL DEPENDS ${PIXEL_SHADER_OUTPUTS})
    add_executable(PixelWorld engine/src/Main.cpp engine/src/Renderer.cpp engine/src/Audio.cpp)
    target_include_directories(PixelWorld PRIVATE "${miniaudio_SOURCE_DIR}")
    target_link_libraries(PixelWorld PRIVATE pixel_core pixel_imgui)
    if(WIN32)
        target_link_libraries(PixelWorld PRIVATE winmm)
    endif()
    if(MSVC)
        target_compile_options(PixelWorld PRIVATE /W4 /utf-8)
    endif()
    set_target_properties(PixelWorld PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(PixelWorld PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${CMAKE_BINARY_DIR}/bin")
    endforeach()
    add_dependencies(PixelWorld pixel_shaders)
    if(PIXEL_BUILD_SCRIPTS)
        add_dependencies(PixelWorld pixel_scripts)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(pixel_world_tests tests/world_tests.cpp)
    target_link_libraries(pixel_world_tests PRIVATE pixel_core)
    add_test(NAME terrain_and_physics COMMAND pixel_world_tests)
    add_executable(pixel_storage_tests tests/storage_tests.cpp)
    target_link_libraries(pixel_storage_tests PRIVATE pixel_core)
    add_test(NAME persistence_and_settings COMMAND pixel_storage_tests)
    if(PIXEL_BUILD_SCRIPTS)
        add_executable(pixel_script_tests tests/script_host_tests.cpp)
        target_link_libraries(pixel_script_tests PRIVATE pixel_core)
        add_dependencies(pixel_script_tests pixel_scripts)
        add_test(NAME native_managed_bridge COMMAND pixel_script_tests "${CMAKE_BINARY_DIR}/bin/scripts")
    endif()
endif()
